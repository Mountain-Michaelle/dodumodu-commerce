// The local file server: what Cloudflare Pages and the R2 public domain do in
// production, done with a filesystem. LOCAL ONLY — see shims.ts.
//
// Everything here goes through mediaPath.ts. A request path is untrusted input
// (a crawler, a typo, someone probing), and a filesystem answers ".." without
// complaint, so each request is resolved and then proved to still be inside the
// root before a single byte is read. Refusals answer 404 rather than 403 so a
// probe learns nothing about the layout.
import { readFileSync } from "node:fs";
import { resolve } from "node:path";
import type { Context, Hono } from "hono";
import type { Env } from "../env";
import { contentTypeFor, resolveMediaPath, resolveWithinRoot } from "./mediaPath";

type App = Hono<{ Bindings: Env }>;

/** Hashed bundles are immutable (dist/_headers says the same); everything
 *  else must be revalidated so an edit is visible without a hard refresh. */
function cacheControlFor(pathname: string): string {
  return /\.[a-f0-9]{8}\.min\.(css|js)$/.test(pathname)
    ? "public, max-age=31536000, immutable"
    : "no-cache";
}

function fileResponse(c: Context, filePath: string, cacheControl: string) {
  const buffer = readFileSync(filePath);
  // A view over the buffer rather than a copy — these are image bytes.
  const bytes = new Uint8Array(buffer.buffer, buffer.byteOffset, buffer.byteLength);
  return c.body(bytes, 200, {
    "Content-Type": contentTypeFor(filePath),
    "Cache-Control": cacheControl,
    "X-Content-Type-Options": "nosniff",
  });
}

/**
 * GET /media/<key> — the local stand-in for the R2 public domain. Uploaded
 * files land in MEDIA_DIR and are served from there; IMAGE_DELIVERY_BASE in
 * .env points the frontend's mediaUrl() at this prefix locally.
 */
export function mountMedia(app: App, mediaRoot: string): void {
  const root = resolve(mediaRoot);
  const prefix = "/media/";

  app.get("/media/*", (c) => {
    // The key is taken from the request path rather than a wildcard param: it
    // is the value the client actually asked for, still percent-encoded, which
    // is exactly what the resolver expects (it decodes once, then checks).
    const requested = c.req.path;
    const key = requested.startsWith(prefix) ? requested.slice(prefix.length) : "";
    const file = resolveMediaPath(root, key);
    if (!file) return c.text("Not found", 404);
    return fileResponse(c, file, "no-store");
  });
}

/**
 * GET * — serves the built site from dodumodu_site/dist, so local development
 * is a single origin on a single port: no CORS to configure, no second server
 * to keep running, and the frontend's API_BASE can simply be empty.
 */
export function mountStatic(app: App, distRoot: string, homePath = "/products/glow-serum/"): void {
  const root = resolve(distRoot);

  const serveFile = (c: Context, urlPath: string) => {
    const relative = urlPath.replace(/^\/+/, "");
    const candidates = relative === "" || relative.endsWith("/")
      ? [`${relative}index.html`]
      : [relative, `${relative}/index.html`];

    for (const candidate of candidates) {
      // null means "unusable key" (traversal, bad encoding) or "not found";
      // either way the answer to the client is the same 404.
      const file = resolveMediaPath(root, candidate);
      if (file) return fileResponse(c, file, cacheControlFor(urlPath));
    }
    return c.text("Not found", 404);
  };

  app.get("/", (c) => serveFile(c, homePath));
  // Registered last, so every API route above wins its own path first.
  app.get("*", (c) => {
    const requested = c.req.path;
    // A traversal attempt is rejected before the fallback is even considered.
    if (!resolveWithinRoot(root, requested.replace(/^\/+/, ""))) return c.text("Not found", 404);
    return serveFile(c, requested);
  });
}
