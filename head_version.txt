<!-- =====================================================================
     11. FORM UX FALLBACK (progressive enhancement)
     Keeps +/- quantity, textarea auto-grow, inline validation and the submit
     loading state working even before/without the bundled orchestrator.
     Hand-off is guarded by form[data-enhanced="true"].
     ===================================================================== -->
<script>
(function () {
  var form = document.getElementById('order-form');
  if (!form) return;
  var owned = function () { return form.dataset.enhanced === 'true'; };

  /* ---- Quantity stepper (custom + / -) ---- */
  var stepper = document.getElementById('quantity-stepper');
  var stepperOwned = function () { return stepper && stepper.dataset.enhanced === 'true'; };

  if (stepper) {
    var input = stepper.querySelector('input[name="quantity"]');
    var minus = stepper.querySelector('.quantity-stepper-btn[data-action="decrement"]');
    var plus = stepper.querySelector('.quantity-stepper-btn[data-action="increment"]');
    var sync = function () {
      var max = parseInt(input.max, 10) || 20;
      var value = parseInt(input.value, 10);
      value = isNaN(value) ? 1 : Math.min(max, Math.max(1, value));
      input.value = value;
      if (minus) minus.disabled = value <= 1;
      if (plus) plus.disabled = value >= max;
    };
    stepper.addEventListener('click', function (event) {
      if (stepperOwned()) return;
      var btn = event.target.closest && event.target.closest('.quantity-stepper-btn');
      if (!btn || btn.disabled) return;
      input.value = (parseInt(input.value, 10) || 1) + (btn.dataset.action === 'increment' ? 1 : -1);
      sync();
    });
    input.addEventListener('input', function () { if (!stepperOwned()) sync(); refreshCod(); });
    input.addEventListener('blur', function () { if (!stepperOwned()) sync(); refreshCod(); });
    sync();
  }

  /* ---- COD gating (preview mirror of codRules.js; server always re-checks) ----
     Frontend default only: backend payment-rules replace this via the bundle.
     Never enables what the server would refuse - disables + explains instead. */
  var COD_STATES = ['Lagos', 'Ogun', 'Enugu', 'Abuja'];
  var COD_BULK = 5; // first bulk quantity (backend seed): 5+ pays now
  var codHintEl = document.getElementById('cod-hint');
  var stateSel = form.elements['state'];
  var qtyEl = form.elements['quantity'];

  function normState(v) {
    var s = String(v || '').trim().toLowerCase().replace(/[.,]/g, ' ').replace(/\s+/g, ' ').trim();
    var alias = { 'abuja': 'fct', 'abuja state': 'fct', 'fct abuja': 'fct', 'f c t': 'fct', 'federal capital territory': 'fct', 'lagos state': 'lagos', 'ogun state': 'ogun', 'enugu state': 'enugu' };
    return alias[s] || s;
  }
  function codDenied(state, quantity) {
    var ok = COD_STATES.some(function (x) { return normState(x) === normState(state); });
    if (!ok) return { reason: 'state', message: 'Pay on delivery isn\u2019t available in ' + (state || 'your area') + ' yet \u2014 choose \u201CPay now\u201D and we\u2019ll dispatch today.' };
    if (quantity >= COD_BULK) return { reason: 'bulk', message: 'Pay on delivery covers up to ' + (COD_BULK - 1) + ' items \u2014 choose \u201CPay now\u201D for bulk orders.' };
    return null;
  }
  function refreshCod() {
    if (owned() || !codHintEl) return;
    var q = parseInt(qtyEl && qtyEl.value, 10) || 1;
    var st = stateSel && stateSel.value;
    var denied = codDenied(st, q);
    var codRadio = form.querySelector ? form.querySelector('input[name="paymentMethod"][value="cod"]') : null;
    var codRow = codRadio && codRadio.closest ? codRadio.closest('.radio-option') : null;
    if (denied) {
      if (codRadio && codRadio.checked) {
        var payNow = form.querySelector('input[name="paymentMethod"][value="pay_now"]');
        if (payNow) payNow.checked = true;
      }
      if (codRadio) codRadio.disabled = true;
      if (codRow) codRow.dataset.unavailable = 'true';
      codHintEl.textContent = denied.message;
    } else {
      if (codRadio) codRadio.disabled = false;
      if (codRow) codRow.dataset.unavailable = 'false';
      codHintEl.textContent = 'Pay on delivery is available in your area for up to ' + (COD_BULK - 1) + ' items.';
    }
  }
  if (stateSel) stateSel.addEventListener('change', refreshCod);
  refreshCod();

  /* ---- Draft memory (4 fields only, client-side, 12h expiry) ---- */
  var DRAFT_KEY = 'skincare_order_draft:glow-serum';
  function draftStorage() { try { localStorage.setItem('__d', '1'); localStorage.removeItem('__d'); return localStorage; } catch (e) { return null; } }
  var draftTimer = null;
  function saveDraftSoon() {
    if (owned()) return;
    if (draftTimer) clearTimeout(draftTimer);
    draftTimer = setTimeout(function () {
      var store = draftStorage();
      if (!store) return;
      var d = { savedAt: Date.now() };
      ['customerName', 'phone', 'whatsapp', 'address'].forEach(function (n) {
        var v = form.elements[n] && form.elements[n].value;
        if (v && v.trim()) d[n] = v;
      });
      try { store.setItem(DRAFT_KEY, JSON.stringify(d)); } catch (e) {}
    }, 300);
  }
  (function restoreDraft() {
    var store = null, raw = null;
    try { store = localStorage; raw = store.getItem(DRAFT_KEY); } catch (e) { return; } /* read-only on load; the write probe stays in draftStorage() for saving */
    if (!raw) return;
    try {
      var d = JSON.parse(raw);
      if (!d || (d.savedAt && Date.now() - d.savedAt > 12 * 3600 * 1000)) { try { store.removeItem(DRAFT_KEY); } catch (e2) {} return; }
      ['customerName', 'phone', 'whatsapp', 'address'].forEach(function (n) {
        var el = form.elements[n];
        if (el && !el.value && typeof d[n] === 'string') el.value = d[n];
      });
    } catch (e) {}
  })();
  ['customerName', 'phone', 'whatsapp', 'address'].forEach(function (n) {
    var el = form.elements[n];
    if (el) el.addEventListener('input', saveDraftSoon);
  });

  /* ---- Live price beside the quantity ----
     Preview-only numbers, so the price moves while the customer taps +/- on a
     page that hasn't been built yet: the unit price comes from this page's own
     JSON-LD offer (no second source of truth) and the tiers mirror the seeded
     ones asserted in test/pricing.smoke.mjs. The bundled app replaces both via
     renderLivePrice() and this block stands down on form[data-enhanced]. */
  var priceBox = document.getElementById('quantity-price');
  var priceInput = document.querySelector ? document.querySelector('input[name="quantity"]') : null;

  var PREVIEW_TIERS = [
    { minQuantity: 1, discountPercent: 0, label: null },
    { minQuantity: 2, discountPercent: 10, label: 'Buy 2, save 10%' },
    { minQuantity: 3, discountPercent: 20, label: 'Buy 3 or 4, save 20%' },
    { minQuantity: 5, discountPercent: 25, label: 'Buy 5+, save 25%' }
  ];

  function previewUnitPrice() {
    try {
      var tag = document.querySelector('script[type="application/ld+json"]');
      var nodes = JSON.parse(tag.textContent)['@graph'] || [];
      for (var i = 0; i < nodes.length; i++) {
        if (nodes[i]['@type'] !== 'Product') continue;
        var offer = nodes[i].offers;
        if (offer && offer.length) offer = offer[0];
        var amount = parseFloat(offer && offer.price);
        if (amount > 0) return Math.round(amount * 100); // naira -> kobo
      }
    } catch (err) { /* no price block on the page: the box stays hidden */ }
    return null;
  }

  var unitPriceKobo = previewUnitPrice();

  function nairaText(kobo) {
    return '\u20A6' + Math.round(kobo / 100).toLocaleString('en-NG');
  }

  function activeTier(quantity) {
    var best = null;
    PREVIEW_TIERS.forEach(function (tier) {
      if (tier.minQuantity <= quantity && (!best || tier.minQuantity > best.minQuantity)) best = tier;
    });
    return best;
  }

  function renderPrice() {
    if (!priceBox || !priceInput || unitPriceKobo === null || owned()) return;

    var quantity = parseInt(priceInput.value, 10);
    if (isNaN(quantity) || quantity < 1) quantity = 1;

    var subtotal = unitPriceKobo * quantity;
    var tier = activeTier(quantity);
    var percent = tier ? tier.discountPercent : 0;
    var saved = Math.round(subtotal * percent / 100);

    var parts = [
      '<span class="quantity-price-total">' + nairaText(subtotal - saved) + '</span>',
      '<span class="quantity-price-each">' + nairaText(unitPriceKobo) + ' &times; ' + quantity + '</span>'
    ];

    if (percent > 0) {
      parts.push('<span class="quantity-price-badge">' + percent + '% off &middot; save ' + nairaText(saved) + '</span>');
    }

    /* Every discount, labelled, with the one in force marked - so the customer
       knows what the next tap is worth before making it. */
    var labelled = PREVIEW_TIERS.filter(function (t) { return t.discountPercent > 0 && t.label; });
    if (labelled.length) {
      parts.push('<span class="quantity-price-ladder-title">Every discount</span>');
      parts.push('<ul class="quantity-price-ladder">' + labelled.map(function (t) {
        var active = t === tier;
        return '<li class="quantity-price-tier" data-active="' + active + '"' + (active ? ' aria-current="true"' : '') +
          '><span>' + t.label + '</span><span>' + t.discountPercent + '%</span></li>';
      }).join('') + '</ul>');
    }

    priceBox.innerHTML = parts.join('');
  }

  if (stepper && priceBox && priceInput) {
    /* Registered after the stepper's own handlers, so the value is already
       settled (and clamped) by the time the price is recalculated. */
    stepper.addEventListener('click', renderPrice);
    priceInput.addEventListener('input', renderPrice);
    priceInput.addEventListener('blur', renderPrice);
    renderPrice();
  }

  /* ---- Textareas: no resize grabber, grow with the text ---- */
  var grow = function (el) {
    el.style.height = 'auto';
    el.style.height = Math.max(96, el.scrollHeight) + 'px';
  };
  form.querySelectorAll('textarea').forEach(function (el) {
    el.addEventListener('input', function () { grow(el); });
    if (el.value) grow(el); /* empty: CSS min-height already gives the same 96px, so skip the reflow at load */
  });

  /* ---- Inline validation (same .field / [data-invalid] contract as the app) ---- */
  var rules = {
    customerName: function (v) { return v.trim().length < 2 ? 'Enter your full name.' : ''; },
    phone: function (v) { return /^[0-9+\-\s()]{7,16}$/.test(v.trim()) ? '' : 'Enter a valid phone number.'; },
    whatsapp: function (v) { return /^[0-9+\-\s()]{7,16}$/.test(v.trim()) ? '' : 'Enter a valid WhatsApp number.'; },
    address: function (v) { return v.trim().length < 8 ? 'Add street, area and a landmark.' : ''; },
    quantity: function (v) { return parseInt(v, 10) >= 1 ? '' : 'Quantity must be at least 1.'; },
    state: function (v) { return v ? '' : 'Select your state.'; },
    deliveryDate: function (v) { return v ? '' : 'Choose a delivery date.'; }
  };

  var showError = function (field, message) {
    field.dataset.invalid = message ? 'true' : 'false';
    var box = field.querySelector('.field-error');
    if (box) box.textContent = message;
  };

  form.querySelectorAll('.field').forEach(function (field) {
    var name = field.dataset.field;
    var el = form.elements[name];
    if (!el || !rules[name]) return;
    var check = function () { if (!owned()) showError(field, rules[name](el.value || '')); };
    el.addEventListener('blur', check);
    el.addEventListener('input', function () {
      if (field.dataset.invalid === 'true' || el.value) check();
    });
  });

  /* ---- Submit: validate, then show a real loading state on the button ---- */
  form.addEventListener('submit', function (event) {
    if (owned()) return; // the bundled orchestrator handles submit itself
    event.preventDefault();
    var firstBad = null;

    form.querySelectorAll('.field').forEach(function (field) {
      var name = field.dataset.field;
      var el = form.elements[name];
      if (!el || !rules[name]) return;
      var message = rules[name](el.value || '');
      showError(field, message);
      if (message && !firstBad) firstBad = el;
    });

    var status = form.querySelector('.form-status');
    var button = form.querySelector('.submit-button');

    if (firstBad) {
      if (status) status.textContent = 'Please correct the highlighted fields.';
      firstBad.focus();
      firstBad.scrollIntoView({ behavior: 'smooth', block: 'center' });
      return;
    }

    if (button) { button.dataset.loading = 'true'; button.disabled = true; }
    if (status) status.textContent = 'Placing your order...';

    // No backend in preview mode (the page was opened without `npm run build`):
    // stop the spinner and say so instead of leaving the button stuck.
    setTimeout(function () {
      if (button) { button.dataset.loading = 'false'; button.disabled = false; }
      if (status) status.textContent = 'Preview mode: run `npm run build`, or connect the order API, to complete checkout.';
    }, 900);
  });
})();
</script>

<!-- =====================================================================
     11b. STICKY PREVIEW FALLBACK (progressive enhancement)
     Same pattern as the form fallback above: file:// preview cannot load
     /__SITE_JS__ (placeholder, no hashed bundle), so the bar would never show.
     This mirrors shared/scripts/stick.js with the same show/hide rule and the
     same .is-on class, then stands down the moment the bundle owns the bar
     (data-stick-ready set by stick.js, or form[data-enhanced] from orchestrate).
     Kept dependency-free and rAF-throttled: one rect read per frame max.
     ===================================================================== -->
<script>
(function(){
  var b=document.querySelector('[data-stick]');
  if(!b)return;
  // Do not set data-stick-ready here: the bundled stick.js owns the bar
  // ([data-stick] IS header.site-header now - nothing is cloned).
  var H=document.querySelector(b.getAttribute('data-stick-hide')||''),p=+b.getAttribute('data-stick-hide-px')||50;
  if(p<0||p>400)p=50;
  var S=+b.getAttribute('data-stick-show')||100;
  if(S<0||S>400)S=100;
  var a=document.getElementById('benefits'),shown=false,q=false;
  function read(){
    q=false;var r,near,hidden;
    // .is-on only while within S px of reaching #benefits (scrolling down), and cleared
    // again when scrolling back up past the same S-px threshold - same spot both ways.
    near=a?a.getBoundingClientRect().top<S:false;
    hidden=false;
    if(H){r=H.getBoundingClientRect();hidden=r.top<innerHeight-p&&r.bottom>0;}
    var n=near&&!hidden;
    if(n)document.documentElement.style.setProperty('--stick-h',b.offsetHeight+'px');
    if(n!==shown){shown=n;b.classList.toggle('is-on',n);document.body.classList.toggle('stick-on',n);}
  }
  function t(){if(!q){q=true;(requestAnimationFrame||setTimeout)(read);}}
  window.addEventListener('scroll',function(){t();},{passive:true});
  window.addEventListener('resize',t,{passive:true});
  window.addEventListener('load',()=>setTimeout(read,100));
  window.addEventListener('pageshow',()=>setTimeout(read,100));
})();
</script>
